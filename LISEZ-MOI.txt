┌──────────────────────────────────────────────────────────────────────┐
│  📝 EXEMPLES D'UTILISATION                                            │
└──────────────────────────────────────────────────────────────────────┘

   # Créer un auditeur
   curl -X POST http://localhost/Backend_AGES/Route/api.php/auditeurs \
     -H "Content-Type: application/json" \
     -d '{"matricule":"AUD-001","nom":"Ndiaye","prenom":"Awa"}'

   # Rechercher un auditeur
   curl http://localhost/Backend_AGES/Route/api.php/auditeurs/search/ndiaye

   # Marquer une présence
   curl -X POST http://localhost/Backend_AGES/Route/api.php/presences/mark \
     -H "Content-Type: application/json" \
     -d '{"id_inscription":1,"date_presence":"2024-09-15",
          "statut":"present","id_formateur":1}'

   # Calculer une moyenne
   curl http://localhost/Backend_AGES/Route/api.php/notes/moyenne/1

   # Statistiques de présence
   curl http://localhost/Backend_AGES/Route/api.php/presences/statistiques/1
